// Scratch harness: runs the scenario transcripts through SNAPMAKER's MixedFilamentManager and writes golden hashes+text.
#include "libslic3r/MixedFilament.hpp"
#include "mixed_engine_scenarios.hpp"
#include <fstream>
#include <iostream>
using namespace Slic3r;
int main(int, char **argv)
{
    std::ofstream o(argv[1]);
    o << "// Generated by a scratch harness linking Snapmaker's MixedFilament.cpp (3-snapmaker-main a00dd4fbac) via\n"
         "// MixedFilamentManager::load_custom_entries + apply_gradient_settings + the same transcript() as the test.\n"
         "// Agreement with that code, not correctness. Regenerate with the harness, do not edit by hand.\n#pragma once\n#include <cstdint>\n#include <string>\nnamespace Slic3r::mixed_engine_golden {\nstruct Entry { const char *name; uint64_t hash; size_t length; };\n";
    o << "inline const Entry k_entries[] = {\n";
    for (const auto &s : mixed_engine_scn::scenarios()) {
        MixedFilamentManager::set_auto_generate_enabled(s.rows.find(",o1,") != std::string::npos);
        MixedFilamentManager m;
        std::vector<std::string> cols(s.num_physical, "#FF0000");
        for (size_t i = 0; i < cols.size(); ++i) { static const char *c[] = {"#FF0000","#00FF00","#0000FF","#FFFF00","#FF00FF","#00FFFF"}; cols[i] = c[i % 6]; }
        m.auto_generate(cols);
        m.load_custom_entries(s.rows, cols);
        m.apply_gradient_settings(s.gradient_mode, s.lo, s.hi, s.advanced);
        const std::string t = mixed_engine_scn::transcript(m, s.num_physical);
        uint64_t h = 1469598103934665603ull;
        for (unsigned char c : t) { h ^= c; h *= 1099511628211ull; }
        o << "    { \"" << s.name << "\", " << h << "ull, " << t.size() << " },\n";
        std::ofstream(std::string(argv[2]) + "/" + s.name + ".txt") << t;
    }
    o << "};\n}\n";
}
